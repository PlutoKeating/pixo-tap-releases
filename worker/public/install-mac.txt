#!/bin/bash
# PixoTap macOS 一键安装脚本（bash 3.2 兼容）。
#
#   curl -fsSL https://pixotap.plutokeating.beer/install-mac | bash
#   curl -fsSL https://pixotap.plutokeating.beer/install-mac | bash -s -- --version v1.0.0
#   curl -fsSL https://pixotap.plutokeating.beer/install-mac | bash -s -- --uninstall
#
# 只依赖 curl、shasum、hdiutil、ditto；不使用 sudo。
# 只读查询公开发布仓库 PlutoKeating/pixo-tap-releases 的 GitHub Release（不需要令牌），
# 选取含 mac-universal.dmg 的最新非草稿、非预发布 Release，用同一 Release 的 SHA256SUMS.txt
# 校验 SHA-256，不符立即中止。把 PixoTap.app 装到 /Applications；不可写时改装到 ~/Applications。
# 用户数据不在应用包内，替换与卸载都不会动它。
# 提示：安装包暂未公证，首次打开时 macOS 可能提示「无法验证开发者」，
# 请在 访达 中右键点击 PixoTap 选择「打开」，或到 系统设置 > 隐私与安全性 中点「仍要打开」。
#
# 选项：
#   --version <tag>   安装指定 release（如 v1.0.0）
#   --dry-run         只查询并打印将要执行的动作
#   --uninstall       删除 PixoTap.app（保留用户数据）
#   -h, --help        显示帮助

set -euo pipefail

REPO="PlutoKeating/pixo-tap-releases"
API_URL="https://api.github.com/repos/$REPO/releases?per_page=100"
APP_NAME="PixoTap.app"

WANT_VERSION=""
DRY_RUN=0
DO_UNINSTALL=0
WORK_DIR=""
MOUNT_POINT=""

info() { printf '%s\n' "$*" >&2; }
warn() { printf '提示：%s\n' "$*" >&2; }
die() { printf '错误：%s\n' "$*" >&2; exit 1; }

cleanup() {
  if [ -n "$MOUNT_POINT" ] && [ -d "$MOUNT_POINT" ]; then
    hdiutil detach "$MOUNT_POINT" -quiet >/dev/null 2>&1 || hdiutil detach "$MOUNT_POINT" -force -quiet >/dev/null 2>&1 || true
  fi
  if [ -n "$WORK_DIR" ] && [ -d "$WORK_DIR" ]; then
    rm -rf "$WORK_DIR"
  fi
}
trap cleanup EXIT

usage() {
  sed -n '2,23p' "$0" 2>/dev/null | sed 's/^# \{0,1\}//' || true
}

while [ $# -gt 0 ]; do
  case "$1" in
    --version) [ $# -ge 2 ] || die "--version 需要一个版本号。"; WANT_VERSION="$2"; shift 2 ;;
    --dry-run) DRY_RUN=1; shift ;;
    --uninstall) DO_UNINSTALL=1; shift ;;
    -h|--help) usage; exit 0 ;;
    *) die "未知选项：$1（用 --help 查看用法）" ;;
  esac
done

[ "$(uname -s)" = "Darwin" ] || die "这个脚本只适用于 macOS。Linux 请用 install，Windows 请用 install.ps1。"
[ -n "${HOME:-}" ] || die "没有 HOME 环境变量，无法确定安装位置。"

# 目标目录：优先 /Applications；已存在于 ~/Applications 则沿用；/Applications 不可写则 ~/Applications。
pick_target() {
  if [ -d "$HOME/Applications/$APP_NAME" ] && [ ! -d "/Applications/$APP_NAME" ]; then
    TARGET_DIR="$HOME/Applications"
  elif [ -w "/Applications" ]; then
    TARGET_DIR="/Applications"
  else
    TARGET_DIR="$HOME/Applications"
    warn "/Applications 不可写（不使用 sudo），改装到 $TARGET_DIR。"
  fi
}

quit_running() {
  if pgrep -x "PixoTap" >/dev/null 2>&1; then
    info "正在退出运行中的 PixoTap……"
    osascript -e 'tell application "PixoTap" to quit' >/dev/null 2>&1 || true
    local i=0
    while pgrep -x "PixoTap" >/dev/null 2>&1 && [ $i -lt 15 ]; do sleep 1; i=$((i + 1)); done
    if pgrep -x "PixoTap" >/dev/null 2>&1; then
      die "PixoTap 仍在运行，请手动退出后重试。"
    fi
  fi
}

if [ "$DO_UNINSTALL" = 1 ]; then
  FOUND=""
  for d in "/Applications" "$HOME/Applications"; do
    [ -d "$d/$APP_NAME" ] && FOUND="$FOUND $d/$APP_NAME"
  done
  [ -n "$FOUND" ] || die "没有找到已安装的 PixoTap。"
  for app in $FOUND; do
    if [ "$DRY_RUN" = 1 ]; then
      info "[dry-run] 将退出运行中的 PixoTap 并删除 $app（用户数据不会被删除）。"
    else
      quit_running
      rm -rf "$app" || die "无法删除 $app（权限不足？请在访达中手动移到废纸篓）。"
      info "已删除 $app。你的用户数据没有被删除。"
    fi
  done
  exit 0
fi

command -v curl >/dev/null 2>&1 || die "缺少命令 curl。"
if [ "$DRY_RUN" != 1 ]; then
  for c in shasum hdiutil ditto; do
    command -v "$c" >/dev/null 2>&1 || die "缺少命令 $c。"
  done
fi

WORK_DIR="$(mktemp -d "${TMPDIR:-/tmp}/pixotap-install.XXXXXX")"

# 从 GitHub releases JSON（stdin）提取：tag|draft|prerelease|资产名=下载地址;... 每个 release 一行。
parse_releases() {
  grep -oE '"([^"\\]|\\.)*"|[][{}:,]|[^][{}:,[:space:]"]+' | awk '
    function unq(s) { if (substr(s,1,1) == "\"") { s = substr(s, 2, length(s) - 2) } return s }
    BEGIN { depth = 0 }
    {
      tok = $0
      if (tok == "{" || tok == "[") {
        parent = (depth >= 1 && t[depth] == "{") ? k[depth] : ""
        depth++
        t[depth] = tok; k[depth] = ""; pk[depth] = parent
        ex[depth] = (tok == "{") ? "key" : "val"
        if (depth == 2 && tok == "{") { tag = ""; draft = ""; pre = ""; assets = "" }
        if (depth == 4 && tok == "{" && pk[3] == "assets") { an = ""; au = "" }
        next
      }
      if (tok == "}" || tok == "]") {
        if (depth == 4 && pk[3] == "assets" && an != "" && au != "") { assets = assets an "=" au ";" }
        if (depth == 2 && tok == "}") { print tag "|" draft "|" pre "|" assets }
        if (depth > 0) depth--
        next
      }
      if (tok == ":") { ex[depth] = "val"; next }
      if (tok == ",") { ex[depth] = (t[depth] == "{") ? "key" : "val"; next }
      if (depth < 1) next
      if (t[depth] == "{" && ex[depth] == "key") { k[depth] = unq(tok); ex[depth] = "colon"; next }
      v = unq(tok)
      if (depth == 2) {
        if (k[2] == "tag_name") tag = v
        else if (k[2] == "draft") draft = v
        else if (k[2] == "prerelease") pre = v
      } else if (depth == 4 && pk[3] == "assets") {
        if (k[4] == "name") an = v
        else if (k[4] == "browser_download_url") au = v
      }
    }'
}

asset_url() { # $1 = "a=url;b=url;"  $2 = 资产名
  printf '%s' "$1" | tr ';' '\n' | awk -F= -v n="$2" '$1 == n { sub(/^[^=]*=/, ""); print; exit }'
}

manual_hint() {
  info "手工安装：打开 https://github.com/$REPO/releases ，下载 PixoTap-*-mac-universal.dmg 与 SHA256SUMS.txt，"
  info "用 shasum -a 256 校验后打开 DMG，把 PixoTap 拖到「应用程序」。"
}

BODY="$WORK_DIR/releases.json"
info "正在查询 $REPO 的发布版本……"
if ! CODE="$(curl -sS --connect-timeout 15 --max-time 60 -o "$BODY" -w '%{http_code}' -H 'Accept: application/vnd.github+json' -A 'pixotap-install-mac' "$API_URL")"; then
  info "无法连接 GitHub。请检查网络或代理设置。"
  manual_hint
  exit 1
fi
case "$CODE" in
  200) ;;
  403|429) info "GitHub API 限流或拒绝（HTTP $CODE）：匿名请求每小时有次数上限，稍后重试。"; manual_hint; exit 1 ;;
  404) info "找不到发布仓库 $REPO（HTTP 404），或它还没有任何 release。"; manual_hint; exit 1 ;;
  *) info "GitHub API 返回了意外状态 HTTP $CODE。"; manual_hint; exit 1 ;;
esac

PICKED_TAG=""; DMG_NAME=""; DMG_URL=""; SUMS_URL=""; SIG_URL=""
while IFS= read -r line; do
  [ -n "$line" ] || continue
  IFS='|' read -r tag draft pre assets <<EOT
$line
EOT
  [ -n "$tag" ] || continue
  [ "$draft" != "true" ] || continue
  if [ -n "$WANT_VERSION" ]; then
    [ "$tag" = "$WANT_VERSION" ] || [ "$tag" = "v${WANT_VERSION#v}" ] || continue
  else
    [ "$pre" != "true" ] || continue
  fi
  name="$(printf '%s' "$assets" | tr ';' '\n' | awk -F= '$1 ~ /^PixoTap-.*mac-universal\.dmg$/ { print $1; exit }')"
  [ -n "$name" ] || continue
  PICKED_TAG="$tag"; DMG_NAME="$name"
  DMG_URL="$(asset_url "$assets" "$name")"
  SUMS_URL="$(asset_url "$assets" "SHA256SUMS.txt")"
  SIG_URL="$(asset_url "$assets" "SHA256SUMS.txt.sig")"
  break
done <<EOT
$(parse_releases <"$BODY")
EOT

if [ -z "$PICKED_TAG" ]; then
  if [ -n "$WANT_VERSION" ]; then
    die "在 $REPO 里没有找到版本 $WANT_VERSION 的 macOS 安装包（mac-universal.dmg）。"
  fi
  info "暂无 macOS 发布：$REPO 里还没有包含 mac-universal.dmg 的正式 release。"
  manual_hint
  exit 1
fi
[ -n "$SUMS_URL" ] || die "release $PICKED_TAG 缺少 SHA256SUMS.txt，出于安全不会安装未校验的文件。"
case "$DMG_NAME" in *[!A-Za-z0-9._-]*) die "安装包文件名不合法：$DMG_NAME" ;; esac
case "$DMG_URL" in https://github.com/*) ;; *) die "不安全的下载地址：$DMG_URL" ;; esac
case "$SUMS_URL" in https://github.com/*) ;; *) die "不安全的校验文件地址：$SUMS_URL" ;; esac

info "找到版本 $PICKED_TAG：$DMG_NAME"
if [ -n "$SIG_URL" ]; then
  info "该版本带有已签名清单（SHA256SUMS.txt.sig）。注意：本脚本只校验 SHA-256，未验证签名。"
else
  warn "该版本没有签名清单，仅依赖 SHA-256 校验。"
fi
pick_target
warn "安装包暂未公证；首次打开时 macOS 可能提示「无法验证开发者」，请右键点击 PixoTap 选择「打开」。"

if [ "$DRY_RUN" = 1 ]; then
  info "[dry-run] 将下载：$DMG_URL"
  info "[dry-run] 将下载：$SUMS_URL"
  info "[dry-run] 将用 shasum -a 256 校验，不符则中止。"
  info "[dry-run] 将挂载 DMG，把 $APP_NAME 复制到 $TARGET_DIR（已存在则先退出运行中的实例并替换，用户数据不受影响），再卸载镜像并清理临时文件。"
  exit 0
fi

DMG_FILE="$WORK_DIR/$DMG_NAME"
SUMS_FILE="$WORK_DIR/SHA256SUMS.txt"
info "下载 $DMG_NAME ……"
curl -fsSL --connect-timeout 15 -A 'pixotap-install-mac' -o "$DMG_FILE" "$DMG_URL" || die "下载安装包失败。"
curl -fsSL --connect-timeout 15 -A 'pixotap-install-mac' -o "$SUMS_FILE" "$SUMS_URL" || die "下载校验文件失败。"

EXPECTED="$(awk -v n="$DMG_NAME" '{ f = $2; sub(/^\*/, "", f); if (f == n && length($1) == 64) { print tolower($1); exit } }' "$SUMS_FILE")"
[ -n "$EXPECTED" ] || die "SHA256SUMS.txt 里没有 $DMG_NAME 的校验值，已中止。"
ACTUAL="$(shasum -a 256 "$DMG_FILE" | awk '{ print tolower($1) }')"
[ "$ACTUAL" = "$EXPECTED" ] || die "SHA-256 校验失败，文件可能被篡改或下载不完整；没有安装。"
info "SHA-256 校验通过。"

MOUNT_POINT="$WORK_DIR/mnt"
mkdir -p "$MOUNT_POINT"
hdiutil attach "$DMG_FILE" -nobrowse -readonly -noverify -mountpoint "$MOUNT_POINT" -quiet >/dev/null || { MOUNT_POINT=""; die "挂载 DMG 失败。"; }
[ -d "$MOUNT_POINT/$APP_NAME" ] || die "安装包结构不正确（缺少 $APP_NAME），已中止。"

mkdir -p "$TARGET_DIR"
STAGE="$TARGET_DIR/.PixoTap-install.$$.app"
rm -rf "$STAGE"
ditto "$MOUNT_POINT/$APP_NAME" "$STAGE" || { rm -rf "$STAGE"; die "复制应用失败。"; }
if [ -d "$TARGET_DIR/$APP_NAME" ]; then
  quit_running
  rm -rf "${TARGET_DIR:?}/$APP_NAME" || { rm -rf "$STAGE"; die "无法替换已有的 $TARGET_DIR/$APP_NAME。"; }
fi
mv "$STAGE" "$TARGET_DIR/$APP_NAME" || { rm -rf "$STAGE"; die "无法放置 $TARGET_DIR/$APP_NAME。"; }

info "PixoTap $PICKED_TAG 已安装到 $TARGET_DIR/$APP_NAME。"
info "打开方式：open \"$TARGET_DIR/$APP_NAME\"（首次打开请右键选择「打开」）。"
